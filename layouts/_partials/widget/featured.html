{{- /* 精选文章 widget（迷你卡片列表） 按「分类法（taxonomy）」中的某个词条（默认「精选」）列出文章，按时间倒序， 每条仅渲染「文章标题」的迷你卡片（不显示封面图、日期）。 精选文章来自「精选」词条：优先在 tags 中查找，若 tags 中没有则回退到 categories。 在 config/_default/params.toml 中配置： [[widgets.homepage]] type = "featured" [widgets.homepage.params] title = "精选文章" # 标题文案 taxonomy = "tags" # 分类法：tags / categories term = "精选" # 词条名称 limit = 0 # 显示条数（0 或不填 = 显示全部精选文章） groupBy = "category" # 分组方式：none / category / series / auto icon = "star" # assets/icons 下的图标名 showLink = true # 标题是否链接到该词条归档页 groupBy 说明： - "none" ：不分组，全部按日期倒序（默认旧行为） - "category" ：按「第一个非精选分类」分组，把同类文章排在一起 - "series" ：按 front matter 的 series 分组 - "auto" ：有 series 用 series，否则用分类 没有系列/分类的文章会归入 uncategorizedName（默认「其他」）分组。 */ -}} {{- $context := .Context -}} {{- $taxonomyName := default "tags" .Params.taxonomy -}} {{- $termName := default "精选" .Params.term -}} {{- $limit := default 0 .Params.limit -}} {{- $widgetTitle := default "精选文章" .Params.title -}} {{- $icon := default "star" .Params.icon -}} {{- $showLink := default true .Params.showLink -}} {{- $groupBy := default "category" .Params.groupBy -}} {{- $uncategorized := default "其他" .Params.uncategorizedName -}} {{- /* 查找顺序：配置的分类法优先，其次回退到另一种常见分类法（tags ↔ categories） */ -}} {{- $taxonomies := slice $taxonomyName -}} {{- range slice "tags" "categories" -}} {{- if not (in $taxonomies .) -}} {{- $taxonomies = $taxonomies | append . -}} {{- end -}} {{- end -}} {{- $entry := false -}} {{- range $taxonomies -}} {{- if not $entry -}} {{- with index $context.Site.Taxonomies . -}} {{- range .ByCount -}} {{- if or (eq (lower .Name) (lower $termName)) (and .Page (eq (lower .Page.Title) (lower $termName))) -}} {{- $entry = . -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if $entry -}} {{- $termPage := $entry.Page -}}
{{- /* 图标与标题同一行 */ -}}
{{ partial "helper/icon" $icon }}

{{- if and $showLink $termPage -}} {{ $widgetTitle }} {{- else -}} {{ $widgetTitle }} {{- end -}}

{{- $pages := $termPage.Pages.ByDate.Reverse -}} {{- if gt $limit 0 -}} {{- $pages = first $limit $pages -}} {{- end -}} {{- /* 为每篇文章计算分组键（保持日期倒序的先后顺序，便于「组内文章相邻」） */ -}} {{- $useSeries := or (eq $groupBy "series") (eq $groupBy "auto") -}} {{- $useCategory := or (eq $groupBy "category") (eq $groupBy "auto") -}} {{- $items := slice -}} {{- range $page := $pages -}} {{- $key := "" -}} {{- if $useSeries -}} {{- with $page.Params.series -}} {{- if reflect.IsSlice . -}} {{- $key = printf "%v" (index . 0) -}} {{- else -}} {{- $key = printf "%v" . -}} {{- end -}} {{- end -}} {{- end -}} {{- if and (eq $key "") $useCategory -}} {{- with $page.Params.categories -}} {{- if reflect.IsSlice . -}} {{- range $c := . -}} {{- if and (eq $key "") (ne $c $termName) -}} {{- $key = $c -}} {{- end -}} {{- end -}} {{- else if ne . $termName -}} {{- $key = . -}} {{- end -}} {{- end -}} {{- end -}} {{- if eq $key "" -}} {{- $key = $uncategorized -}} {{- end -}} {{- $items = $items | append (dict "page" $page "key" $key) -}} {{- end -}}
{{- if or (eq $groupBy "none") (not $items) -}} {{- range $page := $pages -}}
{{ $page.Title }}
 {{- end -}} {{- else -}} {{- /* 分组顺序：按该组最早出现（最新文章）的顺序 */ -}} {{- $order := slice -}} {{- range $it := $items -}} {{- if not (in $order $it.key) -}} {{- $order = $order | append $it.key -}} {{- end -}} {{- end -}} {{- range $name := $order -}}
{{ $name }}
{{- range $it := $items -}} {{- if eq $it.key $name -}} {{- $page := $it.page -}}
{{ $page.Title }}
 {{- end -}} {{- end -}}
{{- end -}} {{- end -}}
{{- else -}} {{- warnf "Featured widget: term '%s' not found in taxonomies %v" $termName $taxonomies -}} {{- end -}}
